// Hisse Analiz ve Portföy Terminali - Ana Mantık // Başlangıç verileri veya hafızadan (localStorage) verileri çekme let portfolio = JSON.parse(localStorage.getItem('hisse_portfolio')) || [ { symbol: 'NVDA', shares: 15, cost: 115.00, price: 130.00 }, { symbol: 'PLTR', shares: 50, cost: 24.50, price: 29.80 } ]; // DOM Elementleri const stockForm = document.getElementById('stock-form'); const portfolioTableBody = document.querySelector('#portfolio-table tbody'); const totalPortfolioValueEl = document.getElementById('total-portfolio-value'); const totalPnLEl = document.getElementById('total-pnl'); // Arayüzü Güncelleme Fonksiyonu function renderPortfolio() { portfolioTableBody.innerHTML = ''; let totalValue = 0; let totalCostBasis = 0; portfolio.forEach((item, index) => { const currentVal = item.shares * item.price; const costVal = item.shares * item.cost; const pnl = currentVal - costVal; const pnlPercentage = costVal > 0 ? (pnl / costVal) * 100 : 0; totalValue += currentVal; totalCostBasis += costVal; const row = document.createElement('tr'); row.innerHTML = ` ${item.symbol} ${item.shares} $${item.cost.toFixed(2)} $${item.price.toFixed(2)} $${currentVal.toFixed(2)} $${pnl.toFixed(2)} (${pnlPercentage.toFixed(2)}%) `; portfolioTableBody.appendChild(row); }); const totalPnL = totalValue - totalCostBasis; const totalPnLPercent = totalCostBasis > 0 ? (totalPnL / totalCostBasis) * 100 : 0; totalPortfolioValueEl.textContent = `$${totalValue.toFixed(2)}`; totalPnLEl.textContent = `$${totalPnL.toFixed(2)} (${totalPnLPercent.toFixed(2)}%)`; totalPnLEl.style.color = totalPnL >= 0 ? '#4ec9b0' : '#f14c4c'; // Verileri tarayıcı hafızasına kaydet localStorage.setItem('hisse_portfolio', JSON.stringify(portfolio)); } // Pozisyon Silme Fonksiyonu window.removeStock = function(index) { portfolio.splice(index, 1); renderPortfolio(); }; // Form Gönderildiğinde Yeni Hisse Ekleme stockForm.addEventListener('submit', (e) => { e.preventDefault(); const symbol = document.getElementById('symbol').value.toUpperCase().trim(); const shares = parseFloat(document.getElementById('shares').value); const cost = parseFloat(document.getElementById('cost').value); const price = parseFloat(document.getElementById('price').value); if (!symbol || isNaN(shares) || isNaN(cost) || isNaN(price)) return; // Aynı hisse varsa güncelle, yoksa yeni ekle const existingIndex = portfolio.findIndex(item => item.symbol === symbol); if (existingIndex > -1) { portfolio[existingIndex] = { symbol, shares, cost, price }; } else { portfolio.push({ symbol, shares, cost, price }); } stockForm.reset(); renderPortfolio(); }); // Sayfa yüklendiğinde ilk render renderPortfolio();
